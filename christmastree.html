圣诞树Python代码 🎄


import time

# 圣诞树立体绘制代码
tree_height = 15  # 树的高度
star = "★"       # 树顶星星
leaf = "✿"       # 树叶
trunk = "■"      # 树干

# 打印树顶星星
print(" " * tree_height + star)

# 打印树叶部分
for i in range(tree_height):
    space = " " * (tree_height - i - 1)
    leaves = leaf * (2 * i + 1)
    print(space + leaves)
    time.sleep(0.1)  # 逐行打印动画效果

# 打印树干
for i in range(3):
    space = " " * (tree_height - 2)
    print(space + trunk)
    time.sleep(0.1)

# 打印祝福语
print("\n" + " " * (tree_height - 3) + "Merry Christmas! 🎄")
